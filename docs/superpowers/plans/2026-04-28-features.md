```typescript
// Maps navigator.language prefix → { code, symbol }
const CURRENCY_MAP: Record<string, { code: string; symbol: string }> = {
  "en-US": { code: "USD", symbol: "$" },
  "en-GB": { code: "GBP", symbol: "£" },
  "de":    { code: "EUR", symbol: "€" },
  "fr":    { code: "EUR", symbol: "€" },
  "es":    { code: "EUR", symbol: "€" },
  "it":    { code: "EUR", symbol: "€" },
  "nl":    { code: "EUR", symbol: "€" },
  "pt-BR": { code: "BRL", symbol: "R$" },
  "ja":    { code: "JPY", symbol: "¥" },
  "ko":    { code: "KRW", symbol: "₩" },
  "zh":    { code: "CNY", symbol: "¥" },
  "en-AU": { code: "AUD", symbol: "A$" },
  "en-CA": { code: "CAD", symbol: "CA$" },
};

function detectCurrency(): { code: string; symbol: string } {
  if (typeof navigator === "undefined") return { code: "USD", symbol: "$" };
  const lang = navigator.language;
  return (
    CURRENCY_MAP[lang] ??
    CURRENCY_MAP[lang.split("-")[0]] ??
    { code: "USD", symbol: "$" }
  );
}

// Regional suggested prices keyed by currency code → ingredient keyword → price
const SUGGESTED_PRICES: Record<string, Record<string, number>> = {
  USD: { chicken: 6.0, beef: 8.5, flour: 1.2, milk: 1.8, eggs: 3.5, butter: 4.0, rice: 2.0, pasta: 1.5, onion: 0.8, garlic: 0.6, tomato: 1.5, cheese: 4.5, olive_oil: 6.0 },
  EUR: { chicken: 4.5, beef: 7.0, flour: 0.8, milk: 1.2, eggs: 2.5, butter: 2.8, rice: 1.5, pasta: 1.2, onion: 0.5, garlic: 0.4, tomato: 1.2, cheese: 3.5, olive_oil: 5.0 },
  GBP: { chicken: 4.0, beef: 6.5, flour: 0.7, milk: 1.0, eggs: 2.2, butter: 2.5, rice: 1.3, pasta: 1.0, onion: 0.4, garlic: 0.35, tomato: 1.0, cheese: 3.0, olive_oil: 4.5 },
};

function getSuggestedPrice(itemName: string, currencyCode: string): number | undefined {
  const prices = SUGGESTED_PRICES[currencyCode] ?? SUGGESTED_PRICES["USD"];
  const lower = itemName.toLowerCase();
  for (const [keyword, price] of Object.entries(prices)) {
    if (lower.includes(keyword.replace("_", " ")) || lower.includes(keyword)) return price;
  }
  return undefined;
}
```

```typescript
function printShoppingList(
  grouped: Record<string, ShoppingItem[]>,
  groupKeys: string[],
  notes: string,
  currencySymbol: string,
) {
  const w = window.open("", "_blank");
  if (!w) return;

  const doc = w.document;
  const style = doc.createElement("style");
  style.textContent = "body{font-family:Georgia,serif;max-width:600px;margin:40px auto;color:#1a0e06}h1{font-size:1.8em;margin-bottom:8px}h2{font-size:1.1em;border-bottom:1px solid #ccc;padding-bottom:4px;margin-top:24px}li{line-height:2;list-style:none;padding:2px 0}li::before{content:'☐  ';font-size:1.1em}.notes{margin-top:24px;white-space:pre-wrap;font-size:0.9em;color:#555}.print-btn{padding:8px 16px;background:#c8522a;color:#fff;border:none;border-radius:8px;cursor:pointer;margin-bottom:24px}@media print{.print-btn{display:none}}";

  const btn = doc.createElement("button");
  btn.className = "print-btn";
  btn.textContent = "🖨 Print / Save PDF";
  btn.addEventListener("click", () => w.print());

  const h1 = doc.createElement("h1");
  h1.textContent = "My Shopping List";

  doc.head.appendChild(style);
  doc.body.appendChild(btn);
  doc.body.appendChild(h1);

  for (const groupKey of groupKeys) {
    const groupItems = grouped[groupKey];
    if (!groupItems?.length) continue;
    const h2 = doc.createElement("h2");
    h2.textContent = groupKey;
    doc.body.appendChild(h2);
    const ul = doc.createElement("ul");
    groupItems.forEach(i => {
      const li = doc.createElement("li");
      li.textContent = [i.amount, i.unit, i.name].filter(Boolean).join(" ");
      ul.appendChild(li);
    });
    doc.body.appendChild(ul);
  }

  if (notes.trim()) {
    const h2 = doc.createElement("h2");
    h2.textContent = "Notes";
    const pre = doc.createElement("div");
    pre.className = "notes";
    // Strip HTML tags for plain-text print
    pre.textContent = notes.replace(/<[^>]+>/g, "");
    doc.body.appendChild(h2);
    doc.body.appendChild(pre);
  }
}
```

```tsx
"use client";

import { useState, useRef, useEffect, useCallback } from "react";
import { X, Plus } from "lucide-react";
import { createClient } from "@/lib/supabase/client";

export type FeatureTag = { id: string; name: string; label: string };

interface TagInputProps {
  recipeId: string;
  initialTags: FeatureTag[];
}

export function TagInput({ recipeId, initialTags }: TagInputProps) {
  const [assigned, setAssigned] = useState<FeatureTag[]>(initialTags);
  const [query, setQuery] = useState("");
  const [suggestions, setSuggestions] = useState<FeatureTag[]>([]);
  const [allTags, setAllTags] = useState<FeatureTag[]>([]);
  const [open, setOpen] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);
  const supabase = createClient();

  // Load all existing tags once
  useEffect(() => {
    supabase.from("wc_feature_tags").select("id, name, label").order("label")
      .then(({ data }) => setAllTags(data ?? []));
  }, []);

  // Filter suggestions as user types
  useEffect(() => {
    if (!query.trim()) { setSuggestions([]); return; }
    const assignedIds = new Set(assigned.map((t) => t.id));
    setSuggestions(
      allTags.filter(
        (t) => !assignedIds.has(t.id) && t.label.toLowerCase().includes(query.toLowerCase())
      ).slice(0, 8)
    );
  }, [query, allTags, assigned]);

  const assignTag = useCallback(async (tag: FeatureTag) => {
    setAssigned((prev) => [...prev, tag]);
    setQuery("");
    setSuggestions([]);
    setOpen(false);
    await supabase.from("wc_recipe_feature_tags").insert({
      recipe_id: recipeId,
      tag_id: tag.id,
      assigned_by: (await supabase.auth.getUser()).data.user?.id,
      ai_assigned: false,
    });
  }, [recipeId, supabase]);

  const createAndAssign = useCallback(async () => {
    const name = query.trim().toLowerCase().replace(/\s+/g, "-");
    const label = query.trim();
    if (!name || !label) return;

    // Create tag
    const { data: tag, error } = await supabase
      .from("wc_feature_tags")
      .insert({ name, label, created_by: (await supabase.auth.getUser()).data.user?.id })
      .select("id, name, label")
      .single();

    if (error || !tag) return;
    setAllTags((prev) => [...prev, tag]);
    await assignTag(tag);
  }, [query, supabase, assignTag]);

  const removeTag = useCallback(async (tag: FeatureTag) => {
    setAssigned((prev) => prev.filter((t) => t.id !== tag.id));
    await supabase.from("wc_recipe_feature_tags")
      .delete()
      .eq("recipe_id", recipeId)
      .eq("tag_id", tag.id);
  }, [recipeId, supabase]);

  const exactMatch = allTags.find((t) => t.label.toLowerCase() === query.trim().toLowerCase());

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
      {/* Assigned tag pills */}
      <div style={{ display: "flex", flexWrap: "wrap", gap: 6 }}>
        {assigned.map((tag) => (
          <span key={tag.id} style={{
            display: "inline-flex", alignItems: "center", gap: 5,
            padding: "4px 10px", borderRadius: 99,
            fontSize: "0.72rem", fontWeight: 600,
            background: "rgba(176,125,86,0.15)",
            color: "#C8913A",
            border: "1px solid rgba(176,125,86,0.3)",
          }}>
            {tag.label}
            <button onClick={() => removeTag(tag)} style={{ display: "flex", alignItems: "center", background: "none", border: "none", cursor: "pointer", color: "#8A6A4A", padding: 0 }}>
              <X style={{ width: 10, height: 10 }} />
            </button>
          </span>
        ))}
      </div>

      {/* Input + dropdown */}
      <div style={{ position: "relative" }}>
        <div style={{ display: "flex", gap: 6 }}>
          <input
            ref={inputRef}
            value={query}
            onChange={(e) => { setQuery(e.target.value); setOpen(true); }}
            onFocus={() => setOpen(true)}
            onKeyDown={(e) => {
              if (e.key === "Enter") {
                e.preventDefault();
                if (suggestions.length > 0) assignTag(suggestions[0]);
                else if (query.trim()) createAndAssign();
              }
              if (e.key === "Escape") setOpen(false);
            }}
            placeholder="Add tag… (type to search or create)"
            style={{
              flex: 1, fontSize: "0.8rem", padding: "8px 12px",
              borderRadius: 10, background: "var(--wc-bg-card, #1C1209)",
              border: "1px solid var(--wc-border-default, #3A2416)",
              color: "var(--wc-text, #EFE3CE)", outline: "none",
            }}
          />
          {query.trim() && !exactMatch && (
            <button onClick={createAndAssign} title="Create new tag"
              style={{ display: "flex", alignItems: "center", gap: 5, padding: "8px 12px", borderRadius: 10, fontSize: "0.75rem", fontWeight: 600, background: "rgba(176,125,86,0.12)", border: "1px solid rgba(176,125,86,0.25)", color: "#B07D56", cursor: "pointer", whiteSpace: "nowrap" }}>
              <Plus style={{ width: 11, height: 11 }} /> Create "{query.trim()}"
            </button>
          )}
        </div>

        {open && suggestions.length > 0 && (
          <div style={{
            position: "absolute", top: "calc(100% + 4px)", left: 0, right: 0, zIndex: 100,
            background: "var(--wc-bg-base, #1A0E06)", border: "1px solid var(--wc-border-default, #3A2416)",
            borderRadius: 10, overflow: "hidden", boxShadow: "0 8px 24px rgba(0,0,0,0.5)",
          }}>
            {suggestions.map((tag) => (
```

- [ ] **Step 1: Create the Edge Function**

Create `supabase/functions/auto-tag-recipe/index.ts`:

```typescript
import Anthropic from "npm:@anthropic-ai/sdk";

const client = new Anthropic({ apiKey: Deno.env.get("ANTHROPIC_API_KEY")! });

const TAG_SLUGS = [
  "kids-meals","low-effort","beginner-friendly","meal-prep","low-budget",
  "quick-meals","one-pot","batch-cooking","freezer-friendly","date-night",
  "comfort-food","high-protein","low-calorie","no-cook","5-ingredients",
  "vegetarian-friendly","spicy","crowd-pleaser","breakfast","dessert",
];

const UTENSILS = [
  "pot","pan","oven","blender","food processor","grill","air fryer",
  "slow cooker","instant pot","wok","steamer","microwave",
];

Deno.serve(async (req) => {
  const { recipe_id, title, ingredients, cook_time_minutes, steps } = await req.json();

  const prompt = `You are classifying a recipe for a cooking app. Given the recipe below, return two JSON arrays:
1. "tags": a subset of these slugs that clearly apply: ${TAG_SLUGS.join(", ")}
2. "utensils": a subset of these that the recipe requires: ${UTENSILS.join(", ")}

Only include tags you are confident about. No hedging.

Recipe title: ${title}
Cook time: ${cook_time_minutes ?? "unknown"} minutes
Ingredients: ${(ingredients ?? []).join(", ")}
Steps summary: ${(steps ?? []).slice(0, 5).join(" | ")}

Respond with ONLY valid JSON, no other text: {"tags": [...], "utensils": [...]}`;

  const message = await client.messages.create({
    model: "claude-haiku-4-5-20251001",
    max_tokens: 256,
    messages: [{ role: "user", content: prompt }],
  });

  const text = (message.content[0] as { type: string; text: string }).text;
  let parsed: { tags: string[]; utensils: string[] } = { tags: [], utensils: [] };
  try { parsed = JSON.parse(text); } catch { /* ignore malformed */ }

  const validTags = (parsed.tags ?? []).filter((t) => TAG_SLUGS.includes(t));
  const validUtensils = (parsed.utensils ?? []).filter((u) => UTENSILS.includes(u));

  // Use Supabase service role to write
  const { createClient } = await import("npm:@supabase/supabase-js");
  const supabase = createClient(
    Deno.env.get("SUPABASE_URL")!,
    Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
  );

  // Fetch tag IDs for valid tag slugs
  if (validTags.length > 0) {
    const { data: tags } = await supabase
      .from("wc_feature_tags")
      .select("id, name")
      .in("name", validTags);

    if (tags && tags.length > 0) {
      await supabase.from("wc_recipe_feature_tags").upsert(
        tags.map((tag) => ({
          recipe_id,
          tag_id: tag.id,
          ai_assigned: true,
        })),
        { onConflict: "recipe_id,tag_id" }
      );
    }
  }

  // Update required_utensils on the recipe row
  await supabase
    .from("recipes")
    .update({ required_utensils: validUtensils })
    .eq("id", recipe_id);

  return new Response(JSON.stringify({ tags: validTags, utensils: validUtensils }), {
    headers: { "Content-Type": "application/json" },
  });
});
```
