# Cuisine & Recipe Cultural Heritage — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add rich cultural heritage content to cuisine pages (Heritage Atlas: history, ingredients, techniques, occasions, cross-culture bridges) and to culturally significant recipe pages (journey banner + heritage sidebar), backed by one-time Wikipedia data + hand-crafted structured fields.

**Architecture:** Extend `CuisineInfo` with 6 new cultural fields; pull Wikipedia summaries once via a dev script committed as `cuisine-wikipedia-cache.ts`; add 3 columns to the `recipes` table; build 3 purely-presentational components wired into the cuisine and recipe detail pages; run a one-time cascade script that auto-marks culturally significant recipes via title matching.

**Tech Stack:** Next.js 16 App Router, TypeScript, Tailwind CSS, Supabase (PostgreSQL), Jest + React Testing Library, tsx (for one-time scripts), Wikipedia REST API v1 (one-time fetch only)

### Task 3: Wikipedia scraper + cache file

- [ ] **Step 1: Write the scraper script**

```ts
// scripts/pull-cuisine-wikipedia.ts
// Run once: npx tsx scripts/pull-cuisine-wikipedia.ts
// Writes src/lib/cuisine-wikipedia-cache.ts

import { writeFileSync } from "fs";
import { join } from "path";
import { CUISINES } from "../src/lib/cuisines";

interface WikiSummary {
  extract: string;
  content_urls: { desktop: { page: string } };
}

async function fetchWikiSummary(title: string): Promise<WikiSummary | null> {
  const url = `https://en.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(title)}`;
  const res = await fetch(url, {
    headers: { "User-Agent": "WhatsCooking/1.0 (recipe app; educational use)" },
  });
  if (!res.ok) {
    console.warn(`  ✗ ${title}: HTTP ${res.status}`);
    return null;
  }
  return res.json() as Promise<WikiSummary>;
}

function truncate(text: string, maxChars = 600): string {
  if (text.length <= maxChars) return text;
  return text.slice(0, text.lastIndexOf(" ", maxChars)) + "…";
}

async function main() {
  const cache: Record<string, { extract: string; source: string }> = {};

  for (const cuisine of CUISINES) {
    if (!cuisine.wikiTitle) continue;
    console.log(`Fetching: ${cuisine.wikiTitle}`);
    const data = await fetchWikiSummary(cuisine.wikiTitle);
    if (!data) continue;
    cache[cuisine.slug] = {
      extract: truncate(data.extract),
      source: data.content_urls.desktop.page,
    };
    console.log(`  ✓ ${cuisine.slug} (${data.extract.length} chars)`);
    // Polite delay to avoid rate-limiting
    await new Promise((r) => setTimeout(r, 300));
  }

  const lines = [
    "// AUTO-GENERATED by scripts/pull-cuisine-wikipedia.ts — do not edit manually",
    "// Re-run the script if cuisine list changes",
    "",
    "export const cuisineWikipediaCache: Record<string, { extract: string; source: string }> = {",
    ...Object.entries(cache).map(
      ([slug, { extract, source }]) =>
        `  ${JSON.stringify(slug)}: {\n    extract: ${JSON.stringify(extract)},\n    source: ${JSON.stringify(source)},\n  },`
    ),
    "};",
  ];

  const outPath = join(__dirname, "../src/lib/cuisine-wikipedia-cache.ts");
  writeFileSync(outPath, lines.join("\n") + "\n", "utf8");
  console.log(`\nWrote ${Object.keys(cache).length} entries to cuisine-wikipedia-cache.ts`);
}

main().catch(console.error);
```

// src/components/cuisine/heritage-atlas-section.tsx
import type { CuisineInfo } from "@/lib/cuisines";

interface Props {
  cuisine: CuisineInfo;
  wikiExtract?: string;
  wikiSource?: string;
}

export function HeritageAtlasSection({ cuisine, wikiExtract, wikiSource }: Props) {
  const hasCulturalContent =
    (cuisine.crossCultureBridges?.length ?? 0) > 0 ||
    (cuisine.stapleIngredients?.length ?? 0) > 0 ||
    wikiExtract ||
    cuisine.historyExtract;

  if (!hasCulturalContent) return null;

  const extract = wikiExtract ?? cuisine.historyExtract;
  const source = wikiSource ?? cuisine.historySource;

  return (
    <div
      className="rounded-xl border p-5 mb-6"
      style={{ background: "#0d0b1e", borderColor: "#2d2b55" }}
    >
      {/* Header */}
      <div className="flex items-center gap-3 mb-5">
        <span
          className="text-xs font-bold tracking-widest"
          style={{ color: "#a78bfa" }}
        >
          HERITAGE ATLAS
        </span>
        <div className="flex-1 h-px" style={{ background: "#2d2b55" }} />
      </div>

      {/* History extract */}
      {extract && (
        <div className="mb-5">
          <p
            className="text-xs font-semibold uppercase tracking-wider mb-2"
            style={{ color: "#f59e0b" }}
          >
            History
          </p>
          <p className="text-sm leading-relaxed" style={{ color: "#94a3b8" }}>
            {extract}
          </p>
          {source && (
            <a
              href={source}
              target="_blank"
              rel="noopener noreferrer"
              className="text-xs mt-1 inline-block hover:underline"
              style={{ color: "#4c4880" }}
            >
              Source: Wikipedia
            </a>
          )}
        </div>
      )}

      {/* Ingredients + Techniques grid */}
      {((cuisine.stapleIngredients?.length ?? 0) > 0 ||
        (cuisine.coreTechniques?.length ?? 0) > 0) && (
        <div className="grid grid-cols-2 gap-4 mb-5">
          {(cuisine.stapleIngredients?.length ?? 0) > 0 && (
            <div>
              <p
                className="text-xs font-semibold uppercase tracking-wider mb-3"
                style={{ color: "#f59e0b" }}
              >
                Staple Ingredients
              </p>
              <div className="flex flex-wrap gap-2">
                {cuisine.stapleIngredients!.map((item) => (
                  <div
                    key={item.name}
                    className="flex items-center gap-2 rounded-lg px-2.5 py-1.5"
                    style={{ background: "#1e293b" }}
                  >
                    <span className="text-base">{item.emoji}</span>
                    <div>
                      <div
                        className="text-xs font-semibold"
                        style={{ color: "#e2e8f0" }}
                      >
                        {item.name}
                      </div>
                      <div className="text-xs" style={{ color: "#64748b" }}>
                        {item.note}
                      </div>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}

          {(cuisine.coreTechniques?.length ?? 0) > 0 && (
            <div>
              <p
                className="text-xs font-semibold uppercase tracking-wider mb-3"
                style={{ color: "#f59e0b" }}
              >
                Core Techniques
              </p>
              <div className="flex flex-wrap gap-2">
                {cuisine.coreTechniques!.map((item) => (
                  <div
                    key={item.name}
                    className="flex items-center gap-2 rounded-lg px-2.5 py-1.5"
                    style={{ background: "#1e293b" }}
                  >
                    <span className="text-base">{item.emoji}</span>
                    <div>
                      <div
                        className="text-xs font-semibold"
                        style={{ color: "#e2e8f0" }}
                      >
                        {item.name}
                      </div>
                      <div className="text-xs" style={{ color: "#64748b" }}>
                        {item.note}
                      </div>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>
      )}

      {/* Cultural occasions */}
      {(cuisine.culturalOccasions?.length ?? 0) > 0 && (
        <div className="mb-5">
          <p
            className="text-xs font-semibold uppercase tracking-wider mb-3"
            style={{ color: "#f59e0b" }}
          >
            Cultural Occasions
          </p>
          <div className="flex flex-wrap gap-2">
            {cuisine.culturalOccasions!.map((occasion) => (
              <span
                key={occasion}
                className="text-xs rounded-full px-3 py-1"
                style={{ background: "#1e293b", color: "#e2e8f0" }}
              >
                {occasion}
              </span>
            ))}
          </div>
        </div>

    </div>

}

```ts
// src/lib/cascade-matching.ts
export function titleMatchesTerms(title: string, terms: string[]): boolean {
  if (terms.length === 0) return false;
  const normalised = title.toLowerCase();
  return terms
    .filter((t) => t.trim().length > 0)
    .some((term) => normalised.includes(term.toLowerCase()));
}
```

```ts
// scripts/cascade-cultural-recipes.ts
// Run once: npx tsx scripts/cascade-cultural-recipes.ts
// Idempotent — safe to re-run.

import { createClient } from "@supabase/supabase-js";
import { CUISINES } from "../src/lib/cuisines";
import { titleMatchesTerms } from "../src/lib/cascade-matching";
import * as dotenv from "dotenv";
import { resolve } from "path";

dotenv.config({ path: resolve(__dirname, "../.env.local") });

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

interface RecipeRow {
  id: string;
  title: string;
  cuisine_type: string | null;
}

async function main() {
  console.log("Fetching all recipes…");
  const { data: recipes, error } = await supabase
    .from("recipes")
    .select("id, title, cuisine_type");

  if (error) throw new Error(`Supabase error: ${error.message}`);
  const rows = (recipes ?? []) as RecipeRow[];
  console.log(`  Found ${rows.length} recipes`);

  let updated = 0;
  let skipped = 0;

  for (const cuisine of CUISINES) {
    if (!cuisine.crossCultureBridges?.length && !cuisine.keyDishes?.length) continue;

    // Collect all match terms for this cuisine
    const terms: string[] = [
      ...(cuisine.keyDishes ?? []),
      ...(cuisine.crossCultureBridges ?? [])
        .map((b) => b.recipeMatch)
        .filter((m): m is string => Boolean(m)),
    ];

    // Build cultural_journey from bridges (use the first matching bridge)
    for (const recipe of rows) {
      if (!titleMatchesTerms(recipe.title, terms)) continue;

      // Find the matching bridge (if any) to build the journey
      const matchingBridge = cuisine.crossCultureBridges?.find((b) =>
        b.recipeMatch
          ? recipe.title.toLowerCase().includes(b.recipeMatch.toLowerCase())
          : false
      );

      let cultural_journey = null;
      if (matchingBridge) {
        // Build a simple 3-4 stop journey from the bridge
        cultural_journey = {
          period: "Historical → present",
          stops: [
            { emoji: "🌍", name: matchingBridge.from.split(" ")[0], note: matchingBridge.from },
            { emoji: "🌉", name: "Fusion", note: "Cultural bridge" },
            { emoji: "🍽️", name: cuisine.name, note: `${cuisine.name} tradition` },
            { emoji: "🌐", name: "Global", note: "Worldwide adoption" },
          ],
        };
      }

      const heritage_notes = matchingBridge
        ? {
            originStory: matchingBridge.story,
            culturalOccasion:
              cuisine.culturalOccasions?.[0]?.replace(/^[^\s]+ /, "") ??
              `Part of ${cuisine.name} culinary tradition`,
            keyIngredientNote: `This dish is a cornerstone of ${cuisine.name} cuisine, reflecting its distinctive character.`,
          }
        : null;

      const { error: updateErr } = await supabase
        .from("recipes")
        .update({
          is_culturally_significant: true,
          cultural_journey,
          heritage_notes,
        })
        .eq("id", recipe.id);

      if (updateErr) {
        console.warn(`  ✗ ${recipe.title}: ${updateErr.message}`);
        skipped++;
      } else {
        console.log(`  ✓ ${recipe.title} [${cuisine.name}]`);
        updated++;
      }
    }
  }

  console.log(`\nDone. Updated: ${updated}, Skipped: ${skipped}`);
}

main().catch(console.error);
```
